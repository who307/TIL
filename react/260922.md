### `async/await`와 예외 처리 (`try...catch`)

```javascript
/**
 * async/await
 * - async 함수 내에서 await 키워드를 사용해 Promise가 완료될 때까지 대기[cite: 40]
 * - 비동기 에러는 try...catch 블록을 통해 동기식 에러 처리와 동일한 구조로 캡처[cite: 40]
 */

function increase(number) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            const result = number + 10;
            if (result > 30) return reject(new Error('숫자가 너무 큽니다.'));[cite: 40]
            resolve(result);[cite: 40]
        }, 1000);
    });
}

async function run() {
    try {
        let result = await increase(0);[cite: 40]
        console.log(result);
        result = await increase(result);[cite: 40]
        console.log(result);
    } catch (error) {
        // Promise reject 발생 시 catch 블록으로 진입[cite: 40]
        console.error(error);[cite: 40]
    }
}

```

### `fetch` API vs `Axios` 라이브러리 비교

```javascript
/**
 * fetch vs Axios
 * 1. fetch: 브라우저 내장. response.ok 확인 및 response.json() 파싱 단계가 필수적임[cite: 41, 42, 43]
 * 2. Axios: 외부 라이브러리. JSON 파싱이 자동화(response.data)되며, 4xx/5xx HTTP 에러를 자동으로 reject함[cite: 43]
 */

const API_URL = 'https://jsonplaceholder.typicode.com/users/1';

// fetch API 활용 방식[cite: 43]
async function callApiWithFetch() {
    try {
        const response = await fetch(API_URL);[cite: 43]
        if (!response.ok) throw new Error('HTTP error'); // ok 여부 직접 확인 필요[cite: 42, 43]
        const data = await response.json();              // json 변환 필요[cite: 41, 43]
        console.log(data);
    } catch (error) {
        console.error('fetch 에러:', error);[cite: 43]
    }
}

// Axios 활용 방식[cite: 43]
async function callApiWithAxios() {
    try {
        const response = await axios.get(API_URL);[cite: 43]
        const data = response.data; // 자동 파싱된 데이터 바로 사용[cite: 43]
        console.log(data);
    } catch (error) {
        // 4xx, 5xx 에러도 catch 블록에서 바로 포착[cite: 43]
        console.error('axios 에러:', error);[cite: 43]
    }
}

```

### React `useEffect` 내 비동기 데이터 통신 및 상태 관리

```jsx
/**
 * API 통신 시 로딩 및 에러 상태 처리 패턴
 * - useEffect 내부에서 비동기 함수를 선언 및 호출[cite: 44, 46]
 * - finally 블록을 활용해 성공/실패와 관계없이 로딩 상태 해제[cite: 45, 46]
 */

const { useState, useEffect } = React;

function UserList() {
    const [users, setUsers] = useState([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
        fetch('https://jsonplaceholder.typicode.com/users')
            .then(res => {
                if (!res.ok) throw new Error('네트워크 응답 오류');[cite: 45]
                return res.json();[cite: 45]
            })
            .then(data => setUsers(data))[cite: 45]
            .catch(err => setError(err))[cite: 45]
            .finally(() => setLoading(false)); // 성공/실패 여부 관계없이 로딩 종료[cite: 45]
    }, []);

    if (loading) return <p>로딩 중...</p>;[cite: 45]
    if (error) return <p>오류 발생: {error.message}</p>;[cite: 45]

    return (
        <ul>
            {users.map(user => (
                <li key={user.id}>{user.name} ({user.company?.name})</li> // 옵셔널 체이닝 사용[cite: 45]
            ))}
        </ul>
    );
}

```

### `Promise.all`을 이용한 다중 상세 API 요청 병렬 처리

```jsx
/**
 * 목록 API 수신 후 각 상세 URL에 대한 추가 API 병렬 요청
 * - Array.prototype.map을 활용해 Promise 객체 배열을 생성[cite: 47]
 * - Promise.all을 통해 모든 요청이 완료될 때까지 한 번에 대기 후 상태 업데이트[cite: 47]
 */

function PokemonList() {
    const [pokemons, setPokemons] = useState([]);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        const fetchPokemons = async () => {
            try {
                const res = await fetch('https://pokeapi.co/api/v2/pokemon');[cite: 47]
                const data = await res.json();[cite: 47]

                // 상세 URL 요청을 위한 Promise 배열 생성[cite: 47]
                const detailPromises = data.results.map(async (poke) => {
                    const detailRes = await fetch(poke.url);[cite: 47]
                    if (!detailRes.ok) throw new Error('포켓몬 정보 불러오기 실패');[cite: 47]
                    return detailRes.json();[cite: 47]
                });

                // 병렬 비동기 요청 전체 수집[cite: 47]
                const results = await Promise.all(detailPromises);[cite: 47]
                setPokemons(results);[cite: 47]
            } catch (err) {
                console.error(err);[cite: 47]
            } finally {
                setLoading(false);[cite: 47]
            }
        };

        fetchPokemons();[cite: 47]
    }, []);

    if (loading) return <p>불러오는 중...</p>;[cite: 47]

    return (
        <div>
            {pokemons.map(pokemon => (
                <div key={pokemon.id}>
                    <img src={pokemon.sprites.front_default} alt={pokemon.name} />[cite: 47]
                    <h3>{pokemon.name}</h3>[cite: 47]
                </div>
            ))}
        </div>
    );
}

```

## 느낀 점

* `.then()` 체이닝 방식보다 `async/await`와 `try...catch` 구문을 조합하는 방식이 코드의 가독성을 대폭 향상시켜 주며, 에러 발생 시 어느 지점에서 예외가 던져졌는지 명확히 파악할 수 있음을 깨달음
* `fetch` API는 HTTP 404, 500 에러에서도 `reject`되지 않으므로 `response.ok`를 별도로 검증해야 하는 반면, `axios`는 HTTP 상태 코드에 따른 예외 처리를 자동으로 제공해 주어 실무에서 생산성을 더욱 높여줄 수 있음을 이해함
* 목록 API 수신 후 각 항목별 상세 정보를 조회해야 하는 케이스에서, 반복문 내부의 직렬 `await` 대신 `map`과 `Promise.all`을 연결하는 방식을 적용하여 네트워크 병목을 획기적으로 줄이는 비동기 최적화 기법을 습득함
