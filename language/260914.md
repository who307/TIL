### 상품 이름 검색 및 `classList` 조작

```javascript
/**
 * 상품 이름 검색 기능
 * 1. 검색어의 공백 제거(trim) 및 소문자 변환(toLowerCase) 수행
 * 2. 빈 검색어 입력 시 alert 경고 및 이전 matched 클래스 일괄 제거
 * 3. for...of 문으로 노드 리스트를 순회하며 includes() 포함 여부에 따라 classList.add / remove 처리
 */

const searchInput = document.getElementById("search-input");
const searchResult = document.getElementById("search-result");
const products = document.querySelectorAll("#product-list .product");

function searchProducts() {
    const currentVal = searchInput.value;
    const processedData = currentVal.trim().toLowerCase();
    let matchedCount = 0;

    searchInput.value = processedData;

    // 빈 검색어 예외 처리
    if (processedData === "") {
        for (const product of products) {
            product.classList.remove("matched");
        }
        return alert("검색어를 입력해주세요.");
    }

    // 조건 검색 및 클래스 토글
    for (const product of products) {
        if (product.textContent.toLowerCase().includes(processedData)) {
            product.classList.add("matched");
            matchedCount++;
        } else {
            product.classList.remove("matched");
        }
    }

    searchResult.textContent = `${matchedCount}개가 검색되었습니다.`;
}

```

### DOM 트래버스 (Parent, Child, Sibling 탐색)

```javascript
/**
 * DOM 트래버스 프로퍼티
 * - parentElement: 직계 부모 요소 접근
 * - firstElementChild / lastElementChild: 첫 번째 및 마지막 자식 요소 접근
 * - previousElementSibling / nextElementSibling: 이전 및 다음 형제 요소 접근
 */

const currentStep = document.getElementById("current-step");
const result = document.getElementById("traversal-result");

function showRelations() {
    // 부모 요소의 스타일 직접 변경
    currentStep.parentElement.style.borderColor = "blue";

    // 부모의 첫 번째 및 마지막 자식 요소에 클래스 추가
    currentStep.parentElement.firstElementChild.classList.add("edge");
    currentStep.parentElement.lastElementChild.classList.add("edge");

    // 이전/다음 형제 요소의 textContent 추출
    const prevText = currentStep.previousElementSibling.textContent;
    const nextText = currentStep.nextElementSibling.textContent;

    result.textContent = `${prevText} ${nextText}`;
}

```

### 동적 DOM 요소 생성 및 삭제

```javascript
/**
 * DOM 요소 동적 생성 및 삭제
 * - document.createElement(): 새 HTML 태그 요소 생성
 * - insertAdjacentElement("beforeend", node): 지정한 위치에 노드 삽입
 * - node.remove(): 해당 노드를 DOM 트리에서 완전히 제거
 */

const todoInput = document.getElementById("todo-input");
const todoList = document.getElementById("todo-list");

function addTodo() {
    const todoInputVal = todoInput.value;

    // 공백 입력 검증
    if (todoInputVal === "" || todoInputVal.trim() === "") {
        return alert("할 일을 입력하세요.");
    }

    // 새 li 요소 생성 및 삽입
    const todoEle = document.createElement("li");
    todoEle.textContent = todoInputVal;
    todoList.insertAdjacentElement("beforeend", todoEle);

    // 입력창 초기화
    todoInput.value = "";
}

function removeLastTodo() {
    // 자식 요소가 존재할 때만 마지막 요소 삭제
    if (todoList.children.length !== 0) {
        todoList.lastElementChild.remove();
    }
}

```

### `dataset` 및 Custom Data Attributes 활용

```javascript
/**
 * dataset & 속성 선택자 조작
 * - dataset.camelCase: HTML의 data-* 속성값을 읽고 수정
 * - setAttribute / removeAttribute: HTML 속성을 직접 추가/삭제
 * - 속성 선택자 [data-product-id="..."]를 활용해 특정 타깃 선택
 */

const products1 = document.querySelectorAll(".product-card");
const selectedResult = document.getElementById("selected-product");

function selectProduct(productId) {
    // 1. 기존 선택된 모든 카드의 클래스 및 dataset 속성 초기화
    for (const prd of products1) {
        prd.classList.remove("selected");
        prd.removeAttribute("data-selected");
    }

    // 2. 속성 선택자로 클릭한 카드 검색
    const selectedProduct = document.querySelector(`.product-card[data-product-id="${productId}"]`);

    // 3. dataset 값 추출
    const selectedProductObj = {
        name: selectedProduct.dataset.productName,
        price: selectedProduct.dataset.price,
    };

    // 4. 선택 상태 클래스 및 속성 부여
    selectedProduct.classList.add("selected");
    selectedProduct.setAttribute("data-selected", "true");

    // 5. 결과 출력
    selectedResult.innerText = `상품명 : ${selectedProductObj.name} \n 상품 가격 : ${selectedProductObj.price}`;
}

```

### Form 입력 연동 및 Class Toggle

```javascript
/**
 * Form 입력 연동 및 테마 전환
 * - color input 값으로 실시간 border-color 변경
 * - select 박스 값에 따른 조건별 classList 조작
 * - classList.toggle(): 클래스가 있으면 제거, 없으면 추가
 * - classList.contains(): 특정 클래스의 존재 여부를 불리언 값으로 확인
 */

const profileNameInput = document.getElementById("profile-name-input");
const profileMessageInput = document.getElementById("profile-message-input");
const profileColorInput = document.getElementById("profile-color-input");
const profileSizeSelect = document.getElementById("profile-size-select");
const profileCard = document.getElementById("profile-card");
const darkModeButton = document.getElementById("dark-mode-button");

function applyProfile() {
    const nameInputVal = profileNameInput.value;
    const messageInputVal = profileMessageInput.value;
    const colorInputVal = profileColorInput.value;

    // 텍스트 및 스타일 적용
    profileCard.children[0].textContent = nameInputVal;
    profileCard.children[1].textContent = messageInputVal;
    profileCard.style.borderColor = colorInputVal;

    // 사이즈 조건 처리
    if (profileSizeSelect.value === "compact") {
        profileCard.classList.add("compact");
    } else {
        profileCard.classList.remove("compact");
    }
}

function toggleDarkMode() {
    // dark 클래스 토글
    profileCard.classList.toggle("dark");

    // 토글 후 현재 다크모드 상태 확인하여 버튼 텍스트 변경
    const isDarkMode = profileCard.classList.contains("dark");
    darkModeButton.textContent = isDarkMode ? "다크 모드 끄기" : "다크 모드 켜기";
}

```

## 느낀 점

* 검색 기능 구현 시 입력값의 `trim()`과 `toLowerCase()` 처리를 먼저 거치고 `includes()`를 활용하면 대소문자 구분을 없애고 공백 예외 처리를 깔끔하게 묶어낼 수 있음을 확인함
* 문서 전체에서 `document.querySelector`로 다시 요소를 찾을 필요 없이 `parentElement`나 `previousElementSibling` 같은 DOM 트래버스 속성을 이용하면 부모와 형제 노드에 손쉽고 직관적으로 접근할 수 있음
* `createElement`로 만든 노드를 `insertAdjacentElement`나 `appendChild`로 원하는 위치에 붙이고 `remove()`로 즉시 삭제하는 동적 DOM 조작 방식을 완전히 습득함
* HTML의 `data-*` 속성과 JavaScript의 `dataset` 객체를 연동하면 UI 요소에 비즈니스 데이터를 자연스럽게 바인딩할 수 있어, 클릭된 요소의 정보 추출 및 상태 관리가 훨씬 용이해짐을 체감함
* `classList.toggle()`과 `classList.contains()`를 조합해 다크모드 상태를 변경함과 동시에 버튼 텍스트를 실시간으로 스위칭하는 깔끔한 토글 UX 구현 패턴을 배움
